Voltus Rail Multi-Die Mode TCL Example

set_distribute_host -local
set_multi_cpu_usage -remote_host 2 -cpu_per_remote_host8 -local_cpu 16 
set DIE1_DEF "../LDIE/DEF/super_filter.def.gz"
set INTERPOSER_DEF "../Interposer/DEF/Interposer.def.gz" 
set_power_def_files   $DIE1_DEF -die_instance_name LDIE1
set_power_def_files   $DIE1_DEF -die_instance_name LDIE2
set_power_def_files   $INTERPOSER_DEF -die_instance_name IDIE 
set LDIE_PGV "../LDIE/PGV/tech_pgv/techonly.cl ../LDIE/PGV/stdcell_pgv/stdcells.cl ../LDIE/PGV/macro_pgv/macros_pll.cl"
set IDIE_PGV "../Interposer/PGV/tech_pgv/techonly.cl" 

set_rail_analysis_config\
-method static \
-enable_xp true \
-die_mode 3dic \
-power_grid_libraries "$LDIE_PGV" \
-die_instance_name LDIE1 \
-accuracy hd \
-tmp_directory_name ./tmp1 \
-limit_number_of_steps false \
-ignore_shorts true \
-enable_mfg_effects true \
-extraction_tech_file ../LDIE/TECH/qrcTechFile \
-temperature 125 

set_rail_analysis_config\
-method static \
-enable_xp true \
-die_mode 3dic \
-power_grid_libraries "$LDIE_PGV" \
-die_instance_name LDIE2 \
-accuracy hd \
-tmp_directory_name ./tmp2 \
-limit_number_of_steps false \
-ignore_shorts true \
-enable_mfg_effects true \
-extraction_tech_file ../LDIE/TECH/qrcTechFile \
-temperature 125

set_rail_analysis_config \
-method static \
-enable_xp true \
-die_mode 3dic \
-power_grid_libraries "$IDIE_PGV" \
-die_instance_name  IDIE \
-accuracy hd \
-tmp_directory_name ./tmp3 \
-limit_number_of_steps false \
-ignore_shorts true \
-enable_mfg_effects true \
-extraction_tech_file ../Interposer/TECH/qrcTechFile \
-tsv_subckt_modelfile_list ../Interposer/data/tsv.ckt \
-temperature 125

set_power_data -reset
set_power_data -format current \[glob "./ldie1_power/static_*.ptiavg"\] -scale 5 -die_instance_name LDIE1
set_power_data -format current \[glob "./ldie2_power/static_*.ptiavg"\] -scale 10 -die_instance_name LDIE2

set_power_pads -format xy -net VDD_AO -file ../LDIE/PADS/VDD_AO.ploc -die_instance_name LDIE1
set_power_pads -format xy -net VSS -file ../LDIE/PADS/VSS.ploc -die_instance_name LDIE1
set_power_pads -format xy -net VDD_AO -file ../LDIE/PADS/VDD_AO.ploc -die_instance_name LDIE2
set_power_pads -format xy -net VSS -file ../LDIE/PADS/VSS.ploc -die_instance_name LDIE2
set_power_pads -format xy -net VDD_AO -file ../Interposer/PLOC/Int_VDD_AO.ploc -die_instance_name IDIE
set_power_pads -format xy -net VSS -file ../Interposer/PLOC/Int_VSS.ploc -die_instance_name IDIE

set_pg_nets -net VDD_AO -force -voltage 1.1 -threshold 1.08 -die_instance_name LDIE1
set_pg_nets -net VSS -force -voltage 0.0 -threshold 0.12 -die_instance_name LDIE1
set_pg_nets -net VDD_AO -force -voltage 1.1 -threshold 1.08 -die_instance_name LDIE2
set_pg_nets -net VSS -force -voltage 0.0 -threshold 0.12 -die_instance_name LDIE2

set_pg_nets -net VDD_AO -force -voltage 1.1 -threshold 1.08 -die_instance_name IDIE
set_pg_nets -net VSS -force -voltage 0.0 -threshold 0.12 -die_instance_name IDIE 

set_die_model -spice_model_file ../DieModel/core_nportModel.sp.subckt -mapping_file ../DieModel/die_model_new.map -die_instance_name LDIE3 
set_multi_die_analysis_mode -analysis_name multi_die -die_list {LDIE1 LDIE2 IDIE} -stacked_die_mapping_file ../StackMap/bump.map 
set_rail_analysis_domain -domain_name ALL \
-power_nets {LDIE2/VDD_AO LDIE1/VDD_AO IDIE/VDD_AO} \
-ground_nets {LDIE2/VSS LDIE1/VSS IDIE/VSS} 
set_package -spice_model_file ../Package/design_single_PinBaseSPICE.ckt -mapping_file ../Package/map.ptmf -sub_circuit_name design_single_PinBaseSPICE
 
set_advanced_rail_options -voltus_rail_include_file_begin vstorm.inc 
report_rail -type domain -output_dir StaticRailResults_test ALL
