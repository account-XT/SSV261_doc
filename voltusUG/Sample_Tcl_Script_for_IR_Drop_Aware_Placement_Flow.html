Sample Tcl Script for IR Drop Aware Placement Flow

The following is the sample Tcl script for IR Drop Aware Placement flow:

#####################################
## IR Drop Analysis - Initial
#####################################
setMultiCpuUsage -localCpu 4
restoreDesign ../design/postRouteOpt_iraware.enc.dat/super_filter

# Power Analysis Environment Setup

set_power_output_dir ./power/power1
set_power_analysis_mode -reset

set_power_analysis_mode \
-analysis_view AV_wc_on \
-disable_static false \
-write_static_currents true \
-binary_db_name dynPower.db \
-create_binary_db true \
-method dynamic_vectorless \
-enable_state_propagation true
set_default_switching_activity -reset
set_default_switching_activity -seq_activity 0.2 -input_activity 0.2 -clock_gates_output 2
set_dynamic_power_simulation -reset
set_dynamic_power_simulation -period 160ns -resolution 20ps

# Dynamic Rail Analysis Environment Setup

set_power_data -reset
set_power_data -format current [ ../*dynamic*.ptiavg ]
set_rail_analysis_mode \
-method dynamic \
-analysis_view AV_wc_on \
-accuracy hd -gif_resolution 0 \
-limit_number_of_steps false \
-power_grid_library ../data/pgv_dir/tech_pgv/techonly.cl \
-enable_rlrp_analysis true
set_pg_nets -net VDD_AO -voltage 0.9 -threshold 0.855
set_pg_nets -net VDD_column -voltage 0.9 -threshold 0.855
set_pg_nets -net VDD_ring -voltage 0.9 -threshold 0.855
set_pg_nets -net VDD_external  -voltage 0.9 -threshold 0.855
set_pg_nets -net VSS -voltage 0 -threshold 0.045
set_rail_analysis_domain -name ALL -pwrnets { VDD_AO VDD_external } -gndnets VSS

# Define Voltage Source Location

set_power_pads -reset
set_power_pads -net VDD_AO -format xy -file ../power_pads/super_filter_VDD_AO.pp
set_power_pads -net VDD_external -format xy -file ../power_pads/super_filter_VDD_external.pp
set_power_pads -net VSS -format xy -file ../power_pads/super_filter_VSS.pp

# Initial Timing Analysis

timeDesign -postRoute -outDir ./timing/timing_initial

#------------------------------------------
# Running RefinePlace - multiple iterations

#------------------------------------------
# Use deleteFiller to delete fillers if there are any fillers in the design to make space for instances to move

setPlaceMode -place_detail_irdrop_aware_effort high -place_detail_preserve_routing true
refinePlace 
mv VoltusDebugIRdropResults VoltusDebugIRdropResults_1
setPlaceMode -place_detail_preserve_routing true
refinePlacemv VoltusDebugIRdropResults VoltusDebugIRdropResults_2
setPlaceMode -place_detail_preserve_routing true
refinePlacemv VoltusDebugIRdropResults VoltusDebugIRdropResults_3
analyze_rail -output ./VoltusRailAnalysisResults -type domain ALL

# Rerouting broken signal nets
ecoRoute

# Final Timing Analysis

timeDesign -postRoute -outDir timing/final
timeDesign -postRoute -outDir timing/final_hold -hold
saveDesign databases/final_database.enc
