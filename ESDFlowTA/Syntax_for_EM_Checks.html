Syntax for EM Checks

rule rule_name
type em
setup
  [-ahdl_include Verilog-A_file]
  [-spice_include SPICE_file]
  [-iv_lib {true | false}]
  [-cross_domain {true | false}]
source
  -type {power | ground}
  -nets net1 net2 net3...
  -short_bumps {true | false}
  -report_threshold value
sink
  -type {power | ground}
  -nets net1 net2 net3...
  -short_bumps {true | false}
  -report_threshold value
clamp_cells cell1 cell2 cell3
clamp_types type1 type2 type3...
nets
  -from list_of_nets|all
  -to list_of_nets|all
bumps
  –from list_of_bumps
  –to list_of_bumps
current value
end rule
