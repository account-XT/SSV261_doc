Setting Up MSV/CPF-Based Design in DMMMC Mode

In this flow, the common power format (CPF) file should not contain any timing-related information. All the library data is present in the viewdefinition.tcl file.

Script -1 (master.tcl):

Same as in non-MSV DMMMC flow.

Script -2 (loadDesign.tcl):

The following points should be noted while migrating to this flow:

  1. All the library-related data and operating conditions are defined in the viewdefinition.tcl file.
  2. Power domains are created in the power.cpf file. This file should not contain any timing or operating conditions-related information.
  3. Timing data and PVT conditions are linked to power domains, by using the update_delay_corner –power_domain command in the viewdefinition.tcl file.
  4. Before reading the CPF, the update_delay_corner command can be used to:
    a) Create place-holder domains.
    b) Populate domains with appropriate MSV data during CPF loading.

Sample design_load.tcl:

####################### Script 2 : "loadDesign.tcl" #########################
##### View definition file does not have LEFs.
read_physical -lef {  ../../libs/lef/FreePDK45_lib_v1.0.lef   \
../../libs/MACRO/LEF/pllclk.lef  \
../../libs/MACRO/LEF/ram_256X16A.lef  \
../../libs/MACRO/LEF/rom_512x16A.lef}
read_mmmc viewDefinition.tcl
set_interactive_constraint_modes functional_mode
# MANUALLY TRANSLATE (WARN-12): Argument '-ignore_undefined_cell' for command 'set_top_module' is obsolete in common UI and is removed.
set_db init_read_netlist_allow_undefined_cells false
read_netlist "../../design/dtmf_recvr_core.v.gz" -top dtmf_recvr_core
init_design
read_def "../../design/full.def"
########################## viewdefinition.tcl ############################
##################################################
# Define the available libraries
##################################################
create_library_set -name libset_slow\
    -timing\
[list ../../libs/liberty/FreePDK45_lib_v1.0_worst.lib\
../../libs/MACRO/LIBERTY/pllclk.lib\
../../libs/MACRO/LIBERTY/ram_256x16A.lib\
../../libs/MACRO/LIBERTY/rom_512x16A.lib]
create_library_set -name libset_fast\
    -timing\
[list ../../libs/liberty/FreePDK45_lib_v1.0_typical.lib\
../../libs/MACRO/LIBERTY/pllclk.lib\
../../libs/MACRO/LIBERTY/ram_256x16A.lib\
../../libs/MACRO/LIBERTY/rom_512x16A.lib]
##################################################
# Define the available PVT corners
##################################################
create_rc_corner -name corner_worst_RCMAX\
    -pre_route_res 1.2\
    -post_route_res {1.2 1.2 1.2}\
    -pre_route_cap 1.2\
    -post_route_cap {1.2 1.2 1.2}\
    -post_route_cross_cap {1.2 1.2 1.2}\
    -pre_route_clock_res 0\
    -pre_route_clock_cap 0
create_rc_corner -name corner_worst_RCMIN\
    -pre_route_res 0.8\
    -post_route_res {0.8 0.8 0.8}\
    -pre_route_cap 0.8\
    -post_route_cap {0.8 0.8 0.8}\
    -post_route_cross_cap {0.8 0.8 0.8}\
    -pre_route_clock_res 0\
    -pre_route_clock_cap 0
##################################################
#Create the timing conditions
#################################################
create_timing_condition -name default_mapping_tc_2\
    -library_sets [list libset_fast]
create_timing_condition -name default_mapping_tc_1\
    -library_sets [list libset_slow]
##################################################
# Create the possible delay corners
# These are combinations of timing conditions and corners
##################################################
create_delay_corner -name dly_crnr_slow_RCMAX\
    -timing_condition {default_mapping_tc_1}\
    -rc_corner corner_worst_RCMAX
create_delay_corner -name delay_corner_fast_RCMAX\
    -timing_condition {default_mapping_tc_2}\
    -rc_corner corner_worst_RCMAX
create_delay_corner -name delay_corner_slow_RCMIN\
    -timing_condition {default_mapping_tc_1}\
    -rc_corner corner_worst_RCMIN
create_delay_corner -name delay_corner_fast_RCMIN\
    -timing_condition {default_mapping_tc_2}\
    -rc_corner corner_worst_RCMIN
##################################################
# Define the available constraint types
##################################################
create_constraint_mode -name functional_mode -sdc_files [list ../../design/dtmf_recvr_core.pr.sdc]
create_constraint_mode -name scan_mode       -sdc_files [list ../../design/dtmf_recvr_core.scan.sdc]

##################################################
# Create the possbile views
# These are combinations of constraints, and delay corners
##################################################
create_analysis_view -name func_slow_RCMAX -constraint_mode functional_mode -delay_corner dly_crnr_slow_RCMAX
create_analysis_view -name func_fast_RCMAX -constraint_mode functional_mode -delay_corner delay_corner_fast_RCMAX
create_analysis_view -name func_slow_RCMIN -constraint_mode functional_mode -delay_corner delay_corner_slow_RCMIN
create_analysis_view -name func_fast_RCMIN -constraint_mode functional_mode -delay_corner delay_corner_fast_RCMIN
create_analysis_view -name scan_slow_RCMAX -constraint_mode scan_mode -delay_corner dly_crnr_slow_RCMAX
create_analysis_view -name scan_fast_RCMAX -constraint_mode scan_mode -delay_corner delay_corner_fast_RCMAX
create_analysis_view -name scan_slow_RCMIN -constraint_mode scan_mode -delay_corner delay_corner_slow_RCMIN
create_analysis_view -name scan_fast_RCMIN -constraint_mode scan_mode -delay_corner delay_corner_fast_RCMIN
##################################################
# Turn on the views that you want to use.
# Not all created views need to be used every time.
##################################################
set_analysis_view -setup [list \
func_slow_RCMAX \
func_fast_RCMAX \
func_slow_RCMIN \
func_fast_RCMIN \
scan_slow_RCMAX \
scan_fast_RCMAX \
scan_slow_RCMIN \
scan_fast_RCMIN \
] -hold [list \
func_slow_RCMAX \
func_fast_RCMAX \
func_slow_RCMIN \
func_fast_RCMIN \
scan_slow_RCMAX \
scan_fast_RCMAX \
scan_slow_RCMIN \
scan_fast_RCMIN \
]

Script - 3 (sta_dmmmc.tcl):

This script contains all the SPEF reading, analysis, and reporting commands.

####################### Script 3: "sta_dmmmc.tcl" ##########################
read_spef -rc_corner corner_worst_RCMAX ../../design/SPEF/corner_worst_RCMAX.spef.gz
read_spef -rc_corner corner_worst_RCMIN ../../design/SPEF/corner_worst_RCMIN.spef.gz
set_db delaycal_enable_si true
################################
# Setup threading
################################
set_multi_cpu_usage -local_cpu 2     ;# Not using multiple machines
################################
# Setup some global timer variables
################################
set_db delaycal_enable_si true ;# Turn on SI analysis
set_db si_glitch_enable_report   true ;# Turn on glitch analysis
# set_global timing_write_sdf_allow_negative_setuphold_sum true ;# Enable SDF to have negative delays
################################
# Select report formatting
################################
set_db timing_report_precision 3
set_table_style -name report_timing -max_widths {70}
set_db timing_report_fields {hpin edge cell transition  load delay incr_delay arrival required}
################################
# set timing derates here if you have them
################################
set_timing_derate -delay_corner dly_crnr_slow_RCMAX -cell_delay -data -early 0.82
set_timing_derate -delay_corner dly_crnr_slow_RCMAX -cell_delay -data -late 1.01
set_timing_derate -delay_corner dly_crnr_slow_RCMAX -cell_delay -clock -early 0.9
set_timing_derate -delay_corner dly_crnr_slow_RCMAX -cell_delay -clock -late 1.1
set_timing_derate -delay_corner dly_crnr_slow_RCMAX -net_delay  -data  -early 0.82
set_timing_derate -delay_corner dly_crnr_slow_RCMAX -net_delay  -data  -late  1.01
set_timing_derate -delay_corner dly_crnr_slow_RCMAX -net_delay  -clock -early 0.9
set_timing_derate -delay_corner dly_crnr_slow_RCMAX -net_delay  -clock -late  1.1
update_timing -full
report_constraint -all_violators  > rep.vio.gz
report_analysis_coverage > rep.coverage
report_timing -path_type endpoint -max_slack 0 > setup.rpt
report_timing -path_type endpoint -max_slack 0 -early > hold.rpt
# Merge these reports later in the main script
report_timing -output_format gtd -max_path 10 -max_slack 10 >setup_maxpath_machine.rpt
report_timing -output_format gtd -nworst 1 > setup_nworst_machine.rpt
report_timing -output_format gtd -max_path 10 -max_slack 10 > hold_maxpath_machine.rpt
report_timing -output_format gtd -nworst 1 > hold_nworst_machine.rpt
